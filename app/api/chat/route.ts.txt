
import { NextResponse } from 'next/server';

export async function POST(req: Request) {
  try {
    const { message } = await req.json();

    // Pemanggilan Meta Model API (Muse)
    const response = await fetch('https://api.meta.ai/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${process.env.META_API_KEY}`,
      },
      body: JSON.stringify({
        model: 'muse-wonder-1.0',
        messages: [
          {
            role: 'system',
            content: 'Anda adalah EcoPulse AI, asisten intelijen keberlanjutan otomotif mewah dan telemetri lingkungan.',
          },
          { role: 'user', content: message },
        ],
        temperature: 0.7,
      }),
    });

    if (!response.ok) {
      throw new Error(`Meta API error: ${response.statusText}`);
    }

    const data = await response.json();
    const reply = data.choices[0]?.message?.content || 'Maaf, tidak ada respon dari Meta AI.';

    return NextResponse.json({ reply });
  } catch (error: any) {
    console.error('Error handling chat API:', error);
    return NextResponse.json(
      { error: 'Gagal menghubungkan ke Meta Muse API' },
      { status: 500 }
    );
  }
}
